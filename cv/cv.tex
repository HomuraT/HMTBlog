% ================================================================= summary ==
\noindent
Lin Ren is a Ph.D. student at the School of Computer Science and Engineering,
Southeast University, China, advised by
Prof.~\href{https://scholar.google.com/citations?user=hbek3dcAAAAJ}{Guohui Xiao} and
Prof.~\href{https://scholar.google.com/citations?user=1gw3LJQAAAAJ}{Guilin Qi}.
His primary research interest is \textit{Knowledge Representation and Reasoning} (KRR),
with contributions from knowledge graphs, natural language processing, and large
language models. A particular emphasis is on \textit{Virtual Knowledge Graphs} (VKGs),
also known as \textit{Ontology-based Data Access} (OBDA).

\section{Research Interests}

\begin{itemize}[leftmargin=1.3em,itemsep=0.15em,topsep=0.2em,parsep=0pt]
  \item Knowledge Representation and Reasoning (KRR)
  \item Virtual Knowledge Graphs, Ontology-based Data Access (OBDA)
  \item Answer Set Programming (ASP)
  \item Large Language Models
  \item Causal Inference
  \item Information Extraction
\end{itemize}
